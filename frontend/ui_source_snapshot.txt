function formatDate(value?: string | null) {
  if (!value) return '—'

  const date = new Date(value)

  if (Number.isNaN(date.getTime())) {
    return '—'
  }

  return date.toLocaleString()
}

type Rotation = 0 | 90 | 180 | 270

type DigitalView = {
  zoom: number
  panX: number
  panY: number
}

type PTZState = {
  supported: boolean
  onvif_supported: boolean
  ptz_supported: boolean
  pan?: number | null
  tilt?: number | null
  zoom?: number | null
}

const DEFAULT_DIGITAL_VIEW: DigitalView = {
  zoom: 1,
  panX: 0,
  panY: 0,
}

type CameraFormState = CameraPayload & {
  password: string
}

const emptyCameraForm: CameraFormState = {
  name: '',
  main_stream_url: '',
  username: '',
  password: '',
  connection_type: 'RTSP',
  description: '',
  location: '',
  enabled: true,
  recording_enabled: true,
  ai_supported: true,
  rtsp_supported: true,
}

function getMainStreamUrl(camera: Camera) {
  return camera.main_stream?.url || ''
}

function Cameras() {
  const token = localStorage.getItem('aks_token') || ''

  const [cameras, setCameras] = useState<Camera[]>([])
  const [loading, setLoading] = useState(true)
  const [actionId, setActionId] = useState<number | null>(null)
  const [message, setMessage] = useState('')
  const [error, setError] = useState('')
  const [streamErrors, setStreamErrors] = useState<
    Record<number, boolean>
  >({})
  const [liveTokens, setLiveTokens] = useState<
    Record<number, string>
  >({})
  const [streamVersion, setStreamVersion] = useState<
    Record<number, number>
  >({})
  const [rotation, setRotation] = useState<
    Record<number, Rotation>
  >({})
  const [fullscreenId, setFullscreenId] = useState<
    number | null
  >(null)
  const [editingCamera, setEditingCamera] = useState<
    Camera | null
  >(null)
  const [showCameraForm, setShowCameraForm] = useState(false)
  const [form, setForm] = useState<CameraFormState>(
    emptyCameraForm,
  )
  const [savingCamera, setSavingCamera] = useState(false)
  const [testingId, setTestingId] = useState<number | null>(null)
  const [testResults, setTestResults] = useState<
    Record<number, string>
  >({})

  const [digitalView, setDigitalView] = useState<
    Record<number, DigitalView>
  >({})

  const [ptzStatus, setPtzStatus] = useState<
    Record<number, PTZState>
  >({})

  const [ptzPresets, setPtzPresets] = useState<
    Record<number, PTZPreset[]>
  >({})

  const [ptzBusy, setPtzBusy] = useState<
    Record<number, boolean>
  >({})

  const previewRefs = useRef<
    Record<number, HTMLDivElement | null>
  >({})

  const dragRefs = useRef<
    Record<
      number,
      {
        startX: number
        startY: number
        panX: number
        panY: number
      }
    >
  >({})

  const clamp = (
    value: number,
    min: number,
    max: number,
  ) => Math.max(min, Math.min(max, value))

  const getDigitalView = (cameraId: number) =>
    digitalView[cameraId] || DEFAULT_DIGITAL_VIEW

  const setCameraDigitalView = (
    cameraId: number,
    next: Partial<DigitalView>,
  ) => {
    setDigitalView((current) => ({
      ...current,
      [cameraId]: {
        ...getDigitalView(cameraId),
        ...next,
      },
    }))
  }

  const zoomIn = (cameraId: number) => {
    const view = getDigitalView(cameraId)

    setCameraDigitalView(cameraId, {
      zoom: clamp(view.zoom + 0.25, 1, 4),
    })
  }

  const zoomOut = (cameraId: number) => {
    const view = getDigitalView(cameraId)

    const zoom = clamp(view.zoom - 0.25, 1, 4)

    setCameraDigitalView(cameraId, {
      zoom,
      panX:
        zoom === 1
          ? 0
          : view.panX,
      panY:
        zoom === 1
          ? 0
          : view.panY,
    })
  }

  const resetDigitalView = (cameraId: number) => {
    setCameraDigitalView(
      cameraId,
      DEFAULT_DIGITAL_VIEW,
    )
  }

  const moveDigital = (
    cameraId: number,
    x: number,
    y: number,
  ) => {
    const view = getDigitalView(cameraId)

    setCameraDigitalView(cameraId, {
      panX: clamp(view.panX + x, -35, 35),
      panY: clamp(view.panY + y, -35, 35),
    })
  }

  const refreshPTZ = useCallback(
    async (cameraId: number) => {
      if (!token) return

      try {
        await detectCameraPTZ(
          token,
          cameraId,
        )

        const status =
          await getCameraPTZStatus(
            token,
            cameraId,
          )

        setPtzStatus((current) => ({
          ...current,
          [cameraId]: status,
        }))

        if (status.ptz_supported) {
          const result =
            await getCameraPTZPresets(
              token,
              cameraId,
            )

          setPtzPresets((current) => ({
            ...current,
            [cameraId]:
              result.presets || [],
          }))
        }
      } catch (err) {
        console.warn(
          `PTZ detection failed for camera ${cameraId}`,
          err,
        )

        setPtzStatus((current) => ({
          ...current,
          [cameraId]: {
            supported: false,
            onvif_supported: false,
            ptz_supported: false,
          },
        }))
      }
    },
    [token],
  )

  const ptzMove = async (
    cameraId: number,
    pan: number,
    tilt: number,
    zoom: number,
  ) => {
    setPtzBusy((current) => ({
      ...current,
      [cameraId]: true,
    }))

    try {
      await moveCameraPTZ(
        token,
        cameraId,
        pan,
        tilt,
        zoom,
        0.8,
      )
    } catch (err: any) {
      setError(
        err?.response?.data?.detail ||
          'PTZ movement failed.',
      )
    } finally {
      setPtzBusy((current) => ({
        ...current,
        [cameraId]: false,
      }))
    }
  }

  const ptzStop = async (
    cameraId: number,
  ) => {
    try {
      await stopCameraPTZ(
        token,
        cameraId,
      )
    } catch (err: any) {
      setError(
        err?.response?.data?.detail ||
          'Unable to stop PTZ.',
      )
    }
  }

  const ptzHome = async (
    cameraId: number,
  ) => {
    try {
      await homeCameraPTZ(
        token,
        cameraId,
      )
    } catch (err: any) {
      setError(
        err?.response?.data?.detail ||
          'Unable to move camera home.',
      )
    }
  }

  const ptzPreset = async (
    cameraId: number,
    presetToken: string,
  ) => {
    try {
      await gotoCameraPTZPreset(
        token,
        cameraId,
        presetToken,
      )
    } catch (err: any) {
      setError(
        err?.response?.data?.detail ||
          'Unable to move to preset.',
      )
    }
  }

  const loadCameras = useCallback(async () => {
    if (!token) return

    try {
      const data = await getCameras(token)
      setCameras(data)
      setError('')

      void Promise.all(
        data.map((camera) =>
          refreshPTZ(camera.id),
        ),
      )
    } catch (err: any) {
      console.error('CAMERA LOAD ERROR:', err)

      if (err?.response?.status === 401) {
        localStorage.removeItem('aks_token')
        window.location.reload()
        return
      }

      setError('Unable to load cameras.')
    } finally {
      setLoading(false)
    }
  }, [token])

  const refreshStatus = useCallback(
    async (cameraList: Camera[]) => {
      if (!token) return

      const updated = await Promise.all(
        cameraList.map(async (camera) => {
          try {
            const status = await getCameraStatus(
              token,
              camera.id,
            )

            return {
              ...camera,
              ...status,
            }
          } catch {
            return camera
          }
        }),
      )

      setCameras((current) =>
        current.map((camera) => {
          const fresh = updated.find(
            (item) => item.id === camera.id,
          )

          return fresh || camera
        }),
      )
    },
    [token],
  )

  const requestLiveToken = useCallback(
    async (cameraId: number) => {
      if (!token) return false

      try {
        const result = await getCameraLiveToken(
          token,
          cameraId,
        )

        setLiveTokens((current) => ({
          ...current,
          [cameraId]: result.token,
        }))

        setStreamErrors((current) => ({
          ...current,
          [cameraId]: false,
        }))

        setStreamVersion((current) => ({
          ...current,
          [cameraId]: (current[cameraId] || 0) + 1,
        }))

        return true
      } catch (err: any) {
        console.error(
          `LIVE TOKEN ERROR [${cameraId}]:`,
          err,
        )

        setLiveTokens((current) => {
          const next = { ...current }
          delete next[cameraId]
          return next
        })

        setStreamErrors((current) => ({
          ...current,
          [cameraId]: true,
        }))

        return false
      }
    },
    [token],
  )

  useEffect(() => {
    loadCameras()
  }, [loadCameras])

  useEffect(() => {
    if (!cameras.length) return

    refreshStatus(cameras)

    const interval = window.setInterval(() => {
      refreshStatus(cameras)
    }, 5000)

    return () => window.clearInterval(interval)
  }, [cameras, refreshStatus])

  useEffect(() => {
    if (!token) return

    cameras.forEach((camera) => {
      const running =
        camera.running === true &&
        camera.status === 'ONLINE'

      if (
        running &&
        !liveTokens[camera.id] &&
        !streamErrors[camera.id]
      ) {
        void requestLiveToken(camera.id)
      }

      if (!running && liveTokens[camera.id]) {
        setLiveTokens((current) => {
          const next = { ...current }
          delete next[camera.id]
          return next
        })
      }
    })
  }, [
    cameras,
    token,
    liveTokens,
    streamErrors,
    requestLiveToken,
  ])

  useEffect(() => {
    const onFullscreenChange = () => {
      const element = document.fullscreenElement

      if (!element) {
        setFullscreenId(null)
        return
      }

      const cameraId = Number(
        (element as HTMLElement).dataset.cameraId,
      )

      setFullscreenId(
        Number.isFinite(cameraId) ? cameraId : null,
      )
    }

    document.addEventListener(
      'fullscreenchange',
      onFullscreenChange,
    )

    return () => {
      document.removeEventListener(
        'fullscreenchange',
        onFullscreenChange,
      )
    }
  }, [])

  const handleStart = async (cameraId: number) => {
    setActionId(cameraId)
    setMessage('')
    setError('')

    try {
      setStreamErrors((current) => ({
        ...current,
        [cameraId]: false,
      }))

      const response = await startCamera(
        token,
        cameraId,
      )

      setMessage(
        `Camera ${cameraId}: ${
          response?.status || 'started'
        }`,
      )

      await loadCameras()
      await requestLiveToken(cameraId)
    } catch (err: any) {
      console.error('START ERROR:', err)

      setError(
        err?.response?.data?.detail ||
          `Unable to start camera ${cameraId}.`,
      )
    } finally {
      setActionId(null)
    }
  }

  const handleStop = async (cameraId: number) => {
    setActionId(cameraId)
    setMessage('')
    setError('')

    try {
      const response = await stopCamera(
        token,
        cameraId,
      )

      setMessage(
        `Camera ${cameraId}: ${
          response?.status || 'stopping'
        }`,
      )

      setLiveTokens((current) => {
        const next = { ...current }
        delete next[cameraId]
        return next
      })

      setStreamErrors((current) => ({
        ...current,
        [cameraId]: false,
      }))

      if (
        document.fullscreenElement ===
        previewRefs.current[cameraId]
      ) {
        await document.exitFullscreen().catch(() => {})
      }

      await loadCameras()
    } catch (err: any) {
      console.error('STOP ERROR:', err)

      setError(
        err?.response?.data?.detail ||
          `Unable to stop camera ${cameraId}.`,

    } finally {
      setActionId(null)

  const handlePreviewWheel = (
    event: React.WheelEvent,
    cameraId: number,
  ) => {
    event.preventDefault()

    const view =
      digitalView[cameraId] ||
      DEFAULT_DIGITAL_VIEW

    const delta =
      event.deltaY < 0
        ? 0.15
        : -0.15

    const zoom = clamp(
      view.zoom + delta,
      1,
      4,
    )

    setCameraDigitalView(
      cameraId,
      {
        zoom,
        panX:
          zoom === 1
            ? 0
            : view.panX,
        panY:
          zoom === 1
            ? 0
            : view.panY,
      },
    )
  }

  const handlePreviewPointerDown = (
    event: React.PointerEvent,
    cameraId: number,
  ) => {
    const view =
      digitalView[cameraId] ||
      DEFAULT_DIGITAL_VIEW

    if (view.zoom <= 1) {
      return
    }

    ;(
      event.currentTarget as HTMLElement
    ).setPointerCapture(event.pointerId)

    dragRefs.current[cameraId] = {
      startX: event.clientX,
      startY: event.clientY,
      panX: view.panX,
      panY: view.panY,
    }
  }

  const handlePreviewPointerMove = (
    event: React.PointerEvent,
    cameraId: number,
  ) => {
    const drag =
      dragRefs.current[cameraId]

    if (!drag) {
      return
    }

    const element =
      previewRefs.current[cameraId]

    if (!element) {
      return
    }

    const rect =
      element.getBoundingClientRect()

    const sensitivity =
      Math.max(
        0.35,
        (digitalView[cameraId]?.zoom || 1) *
          0.6,
      )

    const dx =
      ((event.clientX - drag.startX) /
        Math.max(rect.width, 1)) *
      100 *
      sensitivity

    const dy =
      ((event.clientY - drag.startY) /
        Math.max(rect.height, 1)) *
      100 *
      sensitivity

    setCameraDigitalView(
      cameraId,
      {
        panX: clamp(
          drag.panX + dx,
          -35,
          35,
        ),
        panY: clamp(
          drag.panY + dy,
          -35,
          35,
        ),
      },
    )
  }

  const handlePreviewPointerUp = (
    event: React.PointerEvent,
    cameraId: number,
  ) => {
    delete dragRefs.current[cameraId]

    try {
      ;(
        event.currentTarget as HTMLElement
      ).releasePointerCapture(event.pointerId)
    } catch {
      // Pointer may already have been released.
    }
  }

  const handlePreviewDoubleClick = (
    event: React.MouseEvent,
    cameraId: number,
  ) => {
    event.preventDefault()
    resetDigitalView(cameraId)
  }

  const handleFullscreen = async (
    cameraId: number,
  ) => {
    const element = previewRefs.current[cameraId]

    if (!element) return

    try {
      if (document.fullscreenElement === element) {
        await document.exitFullscreen()
      } else {
        await element.requestFullscreen()
      }
    } catch (err) {
      console.error('FULLSCREEN ERROR:', err)
      setError(
        'Fullscreen is not available in this browser.',
      )
    }
  }

    cameraId: number,
    direction: 1 | -1,
  ) => {
    setRotation((current) => {
      const currentRotation =
        current[cameraId] || 0

      const next =
        (currentRotation + direction * 90 + 360) %
        360

      return {
        ...current,
        [cameraId]: next as Rotation,
      }
    })
  }

    cameraId: number,

    setMessage('')

    setStreamErrors((current) => ({
      ...current,
      [cameraId]: false,

      await requestLiveToken(cameraId)

      setError(

    setMessage('')
    setError('')

      password: '',
      connection_type: camera.connection_type || 'RTSP',
      description: camera.description || '',

      recording_enabled: camera.recording_enabled !== false,
      ai_supported: camera.ai_supported === true,
      rtsp_supported: true,

    setMessage('')
    setError('')

      return 'Camera name is required.'

    if (!form.main_stream_url.trim()) {
      return 'RTSP/stream URL is required.'
    }

    if (
      !/^((rtsp|http|https):\/\/).+/i.test(
        form.main_stream_url.trim(),

    ) {
      return 'Stream URL must start with rtsp://, http://, or https://.'
    }

    return ''

    const streamUrl = form.main_stream_url.trim()
    const payload: CameraPayload = {
      name: form.name.trim(),
      main_stream_url: streamUrl,
      username: form.username.trim(),
      connection_type: form.connection_type,
      description: form.description.trim(),
      location: form.location.trim(),
      enabled: form.enabled,
      recording_enabled: form.recording_enabled,
      ai_supported: form.ai_supported,
      rtsp_supported:
        form.connection_type === 'RTSP' ||
        streamUrl.toLowerCase().startsWith('rtsp://'),
    }

    if (form.password) {
      payload.credential_encrypted = form.password
    }

    return payload
  }

    setError('')

        data?.detail ||

          data?.main_stream_url?.[0] ||
          data?.organization?.[0] ||
          'Unable to save camera.',

    } finally {
      setSavingCamera(false)
    }

        ...current,
        [camera.id]: `${result.status}: ${result.detail}`,

        ...current,
        [camera.id]:
          err?.response?.data?.detail ||
          'Connection test failed.',

      setTestingId(null)

    setError('')

        await stopCamera(token, camera.id)

      setMessage(`${camera.name} removed. Recordings were not deleted.`)
      await loadCameras()
    } catch (err: any) {
      setError(
        err?.response?.data?.detail ||
          'Unable to delete camera.',
      )
    } finally {
      setActionId(null)
    }

function formatBytes(bytes: number) {
  if (!bytes) return '0 B'

  const units = ['B', 'KB', 'MB', 'GB', 'TB']

  let value = bytes
  let unit = 0

  while (
    value >= 1024 &&
    unit < units.length - 1
  ) {
    value /= 1024
    unit += 1
  }

  return `${value.toFixed(unit === 0 ? 0 : 2)} ${units[unit]}`
}

function formatDuration(seconds: number | null) {
  if (
    seconds === null ||
    seconds === undefined
  ) {
    return '--'
  }

  const hours = Math.floor(
    seconds / 3600,
  )

  const minutes = Math.floor(
    (seconds % 3600) / 60,
  )

  const secs = Math.floor(
    seconds % 60,
  )

  if (hours > 0) {
    return `${hours}:${String(minutes).padStart(
      2,
      '0',
    )}:${String(secs).padStart(2, '0')}`
  }

  return `${String(minutes).padStart(
    2,
    '0',
  )}:${String(secs).padStart(2, '0')}`
}

  const [error, setError] = useState('')

  const login = async (e: FormEvent) => {
    e.preventDefault()

    setLoading(true)
    setError('')

    try {
      const response = await axios.post(`${API}/auth/token/`, {
        username,
        password,
      })

      const access = response.data.access

      if (!access) {
        throw new Error('No access token received')
      }

      localStorage.setItem('aks_token', access)
      onLogin(access)
    } catch (err: any) {
      console.error('LOGIN ERROR:', err)

      setError(
        err?.response?.data?.detail ||
          err?.message ||
          'Login failed. Check username and password.',
      )
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="login-page">
      <div className="login-card">
        <div className="brand-mark">AX</div>

        <h1>AKS VisionX</h1>

        <p className="subtitle">
          Intelligent Camera Monitoring & AI Surveillance
        </p>

        <form onSubmit={login}>
          <label>Username</label>

          <input
            value={username}
            onChange={(e) => setUsername(e.target.value)}
            placeholder="Enter username"
            autoComplete="username"
            required
          />

          <label>Password</label>

          <input
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            type="password"
            placeholder="Enter password"
            autoComplete="current-password"
            required
          />

          {error && (
            <div className="alert error">
              {error}
            </div>
          )}

          <button
            className="primary-button"
            disabled={loading}
            type="submit"
          >
            {loading ? 'Signing in...' : 'Sign In'}
          </button>
        </form>

        <div className="login-footer">
          AKS VisionX • Secure Monitoring Platform
        </div>
      </div>
    </div>
  )
}

function App() {
  const [token, setToken] = useState(
    localStorage.getItem('aks_token') || '',
  )

  const logout = () => {
    localStorage.removeItem('aks_token')
    setToken('')
  }

  if (!token) {
    return <Login onLogin={setToken} />
  }

  return (
    <BrowserRouter>
      <Routes>
        <Route
          element={
            <DashboardLayout onLogout={logout} />
          }
        >
          <Route path="/" element={<Overview token={token} />} />
          <Route path="/cameras" element={<Cameras />} />
          <Route path="/recordings" element={<Recordings />} />
          <Route path="/tracking" element={<Tracking />} />
          <Route path="/detections" element={<Detections />} />
          <Route path="/events" element={<Events />} />
          <Route path="/alerts" element={<Alerts />} />
          <Route path="/persons" element={<Persons />} />
          <Route path="/vehicles" element={<Vehicles />} />
          <Route path="/settings" element={<Settings />} />
        </Route>

        <Route
          path="*"
          element={<Navigate to="/" replace />}
        />
      </Routes>
    </BrowserRouter>
  )
}

export default App
